50        gap = np.abs(r @ vals / rn - (vsum - r @ vals) / gn) / vrangegap: ผลต่างค่าเฉลี่ยแดง-เขียวของอายุ/น้ำหนัก/ส่วนสูง/BMI หารด้วยช่วงค่า (normalize)

52        red_dept = r @ dept_ohred_dept = จำนวนคนทีมแดงในแต่ละแผนก
53        split = (((red_dept == 0) | (red_dept == dept_n)) & (dept_n >= 2)).sum(1)  # แผนกกระจุกสีเดียวsplit: นับแผนกที่มี ≥ 2 คนแต่ทั้งแผนกอยู่สีเดียว (ไม่มีแดงเลย หรือแดงหมด)
54        need = max(len(names) - 1, 1)need = จำนวนแผนกที่แต่ละทีมควรมีตัวแทนอย่างน้อย (ขาดได้ไม่เกิน 1 แผนก)
55        div = np.maximum(0, need - (red_dept > 0).sum(1)) + np.maximum(0, need - ((dept_n - red_dept) > 0).sum(1))div: โทษเมื่อทีมแดงหรือทีมเขียวมีตัวแทนแผนกน้อยกว่า need
56        dept_gap = np.abs(2 * red_dept - dept_n).sum(1) / n  # ความต่างแดง-เขียวรวมทุกแผนกdept_gap: ผลรวมความต่าง |แดง−เขียว| ของทุกแผนก หารด้วยจำนวนคนทั้งหมด (ย่อให้สเกลเล็ก)
57        return 3 * gap.sum(1) + 2 * mix + c["dept_weight"] * dept_gap + 10000 * split + 10 * divรวมคะแนนแบบถ่วงน้ำหนัก: gap×3 + mix×2 + dept_weight×dept_gap + split×10000 + div×10

65    def cross(a, b):  # ผสมทีละเพศ ให้จำนวนแดงตรงเป้าเสมอผสมพันธุ์พ่อแม่ a กับ b ให้ลูกหนึ่งตัว
66        x = np.zeros(n, bool)ลูกเริ่มจากทุกคนเป็นเขียว
67        for idx, k in groups:ผสมทีละเพศ เพื่อให้จำนวนแดงของแต่ละเพศถูกต้องเสมอ
68            both = [i for i in idx if a[i] and b[i]]both = คนที่พ่อแม่ทั้งคู่ให้เป็นแดง
69            one = [i for i in idx if a[i] != b[i]]one = คนที่พ่อแม่เห็นไม่ตรงกัน (คนหนึ่งแดง อีกคนเขียว)
70            none = [i for i in idx if not a[i] and not b[i]]none = คนที่พ่อแม่ทั้งคู่ให้เป็นเขียว
71            for p in (both, one, none):วนทั้งสามกลุ่ม
72                rng.shuffle(p)สุ่มลำดับในกลุ่ม เพื่อไม่ให้เลือกคนซ้ำๆ เป็นประจำ

74        return xคืนลูก
